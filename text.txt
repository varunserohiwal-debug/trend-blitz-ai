        let chatMessages = [
            { sender: 'ai', text: 'Namaste Varun! TrendBlitz AI ready hai. Images aur 9:16 Shorts video generation ke liye "Image/Video" tab par click karein.' }

        window.addEventListener('DOMContentLoaded', () => { switchTab('chat'); });

        function switchTab(tab) {
            currentTab = tab;
            ['chat', 'youtube', 'gta', 'image', 'tools', 'profile'].forEach(t => {
                const el = document.getElementById(`nav-${t}`);
                if(el) {
                    if(t === tab) {
                        el.classList.remove('text-slate-400');
                        el.classList.add('text-primaryblue');
                    } else {
                        el.classList.remove('text-primaryblue');
                        el.classList.add('text-slate-400');
                    }
                }
            });

            const viewport = document.getElementById('main-viewport');
            viewport.innerHTML = '';

            if (tab === 'chat') renderChatView(viewport);
            else if (tab === 'youtube') renderYouTubeView(viewport);
            else if (tab === 'gta') renderGTAView(viewport);
            else if (tab === 'image') renderImageView(viewport);
            else if (tab === 'tools') renderToolsView(viewport);
            else if (tab === 'profile') renderProfileView(viewport);
        }

        function renderChatView(container) {
            container.innerHTML = `
                <div class="flex flex-col h-full justify-between space-y-4">
                    <div class="flex items-center justify-between bg-cardbg p-3 rounded-xl border border-bordercolor">
                        <div class="flex items-center space-x-2">
                            <span class="w-2.5 h-2.5 rounded-full bg-green-500 animate-pulse"></span>
                            <span class="text-xs font-semibold text-slate-300">TrendBlitz Assistant Online</span>
                        </div>
                        <span class="text-[10px] px-2 py-0.5 rounded bg-blue-500/20 text-blue-400 border border-blue-500/30">Fast Node</span>
                    </div>
                    
                    <div id="chat-stream" class="flex-1 overflow-y-auto space-y-3 pr-1">
                        ${chatMessages.map(m => `
                            <div class="flex ${m.sender === 'user' ? 'justify-end' : 'justify-start'}">
                                <div class="max-w-[85%] p-3 rounded-2xl text-sm ${m.sender === 'user' ? 'gradient-bg text-white rounded-br-none' : 'bg-cardbg border border-bordercolor text-slate-200 rounded-bl-none'}">
                                    <p>${m.text}</p>
                                </div>
                            </div>
                        `).join('')}
                    </div>

                    <div class="flex items-center space-x-2 bg-cardbg p-2 rounded-xl border border-bordercolor">
                        <input id="chat-input" type="text" placeholder="Ask anything in Hindi/English..." class="flex-1 bg-transparent border-none text-sm text-slate-100 focus:outline-none px-2" onkeypress="if(event.key==='Enter') sendChatMessage()">
                        <button onclick="sendChatMessage()" class="w-9 h-9 rounded-lg gradient-bg flex items-center justify-center text-white shadow-md"><i class="fa-solid fa-paper-plane text-xs"></i></button>
                    </div>
                </div>
            `;
            scrollToBottom('chat-stream');
        }

        function sendChatMessage() {
            const input = document.getElementById('chat-input');
            const val = input.value.trim();
            if(!val) return;
            chatMessages.push({ sender: 'user', text: val });
            input.value = '';
            renderChatView(document.getElementById('main-viewport'));

            setTimeout(() => {
                chatMessages.push({ sender: 'ai', text: `Aapka sawaal "${val}" mil gaya hai! Video shorts aur images generate karne ke liye Image/Video studio use karein.` });
                renderChatView(document.getElementById('main-viewport'));
            }, 500);
        }
